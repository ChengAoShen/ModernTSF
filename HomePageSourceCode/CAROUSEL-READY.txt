   • 精确对齐到中心
   • 确保数据集居中

核心算法
--------------------------------------------------------------------------------

### 循环归一化

function normalizeOffset(offset) {
  const cycleWidth = datasets.length * ITEM_WIDTH;
  let normalized = offset % cycleWidth;
  
  // 保持在合理范围内
  if (normalized < -cycleWidth / 2) normalized += cycleWidth;
  if (normalized > cycleWidth / 2) normalized -= cycleWidth;
  
  return normalized;
}

### 检测中心数据集

function getCurrentDatasetIndex(offset) {
  const normalizedOffset = normalizeOffset(offset);
  const index = Math.round(-normalizedOffset / ITEM_WIDTH);
  return ((index % datasets.length) + datasets.length) % datasets.length;
}

### 动量滚动

let velocity = initialVelocity;
const friction = 0.92;

function animate() {
  velocity *= friction;
  if (Math.abs(velocity) < 0.3) {
    snapToNearest();
    return;
  }
  offset += velocity;
  updateContent();
  requestAnimationFrame(animate);
}

### 平滑吸附

function snapToNearest() {
  const targetOffset = -currentIndex * ITEM_WIDTH;
  function animate() {
    const diff = targetOffset - currentOffset;
    if (Math.abs(diff) < 0.5) return;
    currentOffset += diff * 0.15; // 缓动系数
    requestAnimationFrame(animate);
  }
  animate();
}

技术规格
--------------------------------------------------------------------------------

组件:    dataset-carousel.tsx (10KB)
集成:    leaderboard-enhanced-v3.tsx
框架:    React 19 + TypeScript
动画:    requestAnimationFrame (60fps)
性能:    GPU 加速 (transform)

参数:
  • ITEM_WIDTH: 120px（每个数据集宽度）
  • ITEM_SPACING: 20px（数据集间距）
  • 摩擦系数: 0.92（动量衰减）
  • 吸附系数: 0.15（缓动速度）

效果:
  • 透明度衰减: 1.0 → 0.3
  • 缩放衰减: 1.0 → 0.75
  • 闪光周期: 3秒
  • 模糊半径: 20px
